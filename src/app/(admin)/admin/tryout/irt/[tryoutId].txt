function createColumns(data: any[]): ColumnDef<any>[] {
  if (data.length === 0) return [];
  return Object.keys(data[0]).map((key) => ({
    accessorKey: key,
    header: ({ column }) => {
      return (
        <Button
          variant="ghost"
          onClick={() => column.toggleSorting(column.getIsSorted() === 'asc')}
        >
          {key}
          <ArrowUpDown className="ml-2 h-4 w-4" />
        </Button>
      );
    },
  }));
}

export default function Index() {
  const router = useRouter();
  const { tryoutId } = router.query;
  console.log({ tryoutId });

  const { data: TryoutData } = api.irt.getTryoutDataForIrt.useQuery(
    { tryoutId: tryoutId as string },
    { refetchOnWindowFocus: false },
  );

  console.log({ TryoutData });

  const [csvData, setCSVData] = useState<any[]>([]);
  const [sessionIndex, setSessionIndex] = useState<number>(0);

  const handleDataUploaded = (data: File) => {
    handleGenerate(data);
  };

  const processDataUserAnswer = (sessionIndex: number) => {
    if (!TryoutData) return;
    // const dummyData = data2;
    const dummyData = TryoutData;
    const SessionOne = dummyData.TryoutSession[sessionIndex];
    console.log({ SessionOne });
    const filterData = SessionOne.TryoutSessionParticipant.map(
      (participant, index) => {
        const question = participant.TryoutUserAnswer.map((uAnswer) => {
          const isCorrect = () => {
            if (uAnswer.TryoutAnswers?.value === 5) return 1;
            return 0;
          };
          return {
            q: uAnswer.TryoutQuestion.number,
            correct: isCorrect(),
          };
        }).sort((a, b) => a.q - b.q);
        return {
          p: index + 1,
          question,
        };
      },
    ).map((participant)=>{
      return {
        p: participant.p,
        question: participant.question
      }
    });

    const groupingByParticipant = filterData.reduce(
      (
        acc: {
          [key: string]: {
            q: number;
            correct: number;
          }[];
        },
        item,
      ) => {
        const key = item.p;
        // if (!acc[`p${key}`]) {
        //   acc[`p${key}`] = [];
        // }
        const question = item.question;
        acc[key] = [...question];
        return acc;
      },
      {},
    );

    const groupingArray = Object.keys(groupingByParticipant)
      .map((key) => {
        const question = groupingByParticipant[key];
        const groupingByQuestion = question.reduce(
          (acc: { [key: string]: number }, item) => {
            const key = `q${item.q}`;
            acc[key] = item.correct;
            return acc;
          },
          {},
        );
        if (question.length == 0) {
          return null;
        }
        return {
          p: key,
          ...groupingByQuestion,
        };
      })
      .filter((item) => item);

    console.log({ groupingByParticipant, groupingArray });
    return groupingArray.map((item, index) => {
      return {
        ...item,
        p: `p${index + 1}`,
      };
    });
  };

  const exportData = async (sessionIndex: number) => {
    const fileName = '12313';

    const downloadData = processDataUserAnswer(sessionIndex) || null;

    if (!downloadData) return;

    let wb = utils.book_new(),
      ws = utils.json_to_sheet(downloadData);
    utils.book_append_sheet(wb, ws, 'items');
    writeFile(wb, `${fileName}.csv`);
  };

  const handleGenerate = (file: File) => {
    if (file) {
      Papa.parse(file, {
        header: true,
        skipEmptyLines: true,
        complete: function (results: any) {
          const data: any[] = results.data;
          console.log('data', data);
          setCSVData(data);
        },
        error: function (error: any) {
          console.error(error);
          toaster({
            title: 'Upss',
            condition: 'warning',
            description: 'Gagal membaca file CSV!',
          });
        },
      });
    }
  };

  return (
    <div className="container mx-auto p-4">
      <h1 className="text-3xl font-bold mb-6">SNBT Data Processor</h1>

      <Card className="mb-6">
        <CardHeader>
          <CardTitle>Upload CSV File</CardTitle>
          <div className="flex flex-col gap-2 pt-4">
            <h1 className="font-semibold text-lg">Pilih Sesi</h1>
            <Select
              value={sessionIndex.toString()}
              onValueChange={(value) =>
                value && setSessionIndex(parseInt(value))
              }
            >
              <SelectTrigger className="shadow-md font-medium text-base">
                <SelectValue placeholder="Pilih Sesi" />
              </SelectTrigger>
              <SelectContent>
                {TryoutData?.TryoutSession.map((item, index) => (
                  <SelectItem
                    key={index}
                    value={index.toString()}
                    className="font-medium text-base"

                    {item.TryoutCategory.name} - {item.TryoutSubCategory.name}
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>
        </CardHeader>
        <CardContent className="grid grid-cols-2 gap-4">
          <Card>
            <CardHeader>
              <CardTitle>Download CSV</CardTitle>
            </CardHeader>
            <CardContent className="flex flex-col gap-4">
              <div className="flex gap-4 items-center">
                <Button
                  className="bg-main hover:bg-main-hover w-fit"
                  onClick={() => {
                    exportData(sessionIndex);
                  }}
                >
                  Download
                </Button>
                <p>
                  {TryoutData?.TryoutSession[sessionIndex].TryoutCategory
                    .name || ''}{' '}
                  -{' '}
                  {TryoutData?.TryoutSession[sessionIndex].TryoutSubCategory
                    .name || ''}
                </p>
              </div>
            </CardContent>
          </Card>
          <Card>
            <CardHeader>
              <CardTitle>Upload CSV File</CardTitle>
            </CardHeader>
            <CardContent className="flex flex-col gap-4">
              <CSVUploader onDataUploaded={handleDataUploaded} />
              {csvData.length > 0 && (
                <Button className="w-fit bg-main hover:bg-main-hover">
                  Terapkan 3pl ke Sesi ini
                </Button>
              )}
            </CardContent>
          </Card>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <CardTitle>Preview CSV</CardTitle>
        </CardHeader>
        <CardContent>
          <DataTable columns={createColumns(csvData)} data={csvData} />
        </CardContent>
      </Card>
    </div>
  );
}
